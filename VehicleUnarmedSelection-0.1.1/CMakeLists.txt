cmake_minimum_required(VERSION 3.20)
project(VehicleUnarmedSelection LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Use Windows x86: Visual Studio -A Win32 or i686 LLVM-MinGW")
endif()
add_library(VehicleUnarmedSelection SHARED src/VehicleUnarmedSelection.cpp)
target_compile_features(VehicleUnarmedSelection PRIVATE cxx_std_17)
target_link_libraries(VehicleUnarmedSelection PRIVATE version)
set_target_properties(VehicleUnarmedSelection PROPERTIES PREFIX "" SUFFIX ".asi")
if(MSVC)
  set_property(TARGET VehicleUnarmedSelection PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(VehicleUnarmedSelection PRIVATE /W4 /EHsc)
else()
  target_compile_options(VehicleUnarmedSelection PRIVATE -Wall -Wextra)
  target_link_options(VehicleUnarmedSelection PRIVATE -static)
endif()
